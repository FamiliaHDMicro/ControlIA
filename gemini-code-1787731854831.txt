function Write-Log {
    param([string]$message)
    $timestamp = Get-Date -Format "HH:mm:ss"
    $window.Dispatcher.Invoke([Action]{
        $TxtLog.AppendText("[$timestamp] $message`r`n")
        $TxtLog.ScrollToEnd()
    })
}

function Set-Status {
    param([string]$status)
    $window.Dispatcher.Invoke([Action]{
        $TxtStatus.Text = "Status: $status"
    })
}

# --- LÓGICA DE EXECUÇÃO ---

$BtnLimpeza.Add_Click({
    Set-Status "Executando limpeza..."
    Write-Log "Iniciando limpeza de temporários..."
    
    $tempFolders = @($env:TEMP, "C:\Windows\Temp", "C:\Windows\SoftwareDistribution\Download")
    foreach ($folder in $tempFolders) {
        if (Test-Path $folder) {
            Get-ChildItem -Path $folder -Recurse -Force -ErrorAction SilentlyContinue | 
                Remove-Item -Force -Recurse -ErrorAction SilentlyContinue
        }
    }
    Clear-RecycleBin -Force -ErrorAction SilentlyContinue
    Write-Log "Limpeza concluída com sucesso!"
    Set-Status "Pronto"
})

$BtnModoTurbo.Add_Click({
    Set-Status "Ativando Modo Turbo..."
    try {
        powercfg -setactive e9a42b02-d5df-448d-aa00-03f14749eb61 | Out-Null
        Write-Log "Plano 'Desempenho Máximo' ativado."
    } catch {
        powercfg -setactive 8c5e7fda-e8bf-4a96-9a85-a6e23a8c635c | Out-Null
        Write-Log "Plano 'Alto Desempenho' ativado."
    }
    Set-Status "Pronto"
})

$BtnDNS.Add_Click({
    Set-Status "Limpando cache de rede..."

    Set-Status "Pronto"
})

# Desinstalador Seguro sem Invoke-Expression
$BtnDesinstalar.Add_Click({
    Set-Status "Carregando Desinstalador..."
    
    $paths = @(
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*",
        "HKLM:\Software\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*",
        "HKCU:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*"
    )
    $apps = Get-ItemProperty $paths -ErrorAction SilentlyContinue |
        Where-Object { $_.DisplayName -and $_.SystemComponent -ne 1 -and $_.UninstallString } |
        Select-Object DisplayName, UninstallString | Sort-Object DisplayName

    $form = New-Object System.Windows.Forms.Form
    $form.Text = "Desinstalador Seguro"
    $form.Size = New-Object System.Drawing.Size(600, 500)
    $form.StartPosition = "CenterScreen"

    $listBox = New-Object System.Windows.Forms.CheckedListBox
    $listBox.Location = New-Object System.Drawing.Point(20, 20)
    $listBox.Size = New-Object System.Drawing.Size(540, 380)
    $listBox.CheckOnClick = $true
    foreach ($app in $apps) { [void]$listBox.Items.Add($app.DisplayName) }
    $form.Controls.Add($listBox)

    $btnRun = New-Object System.Windows.Forms.Button
    $btnRun.Text = "Remover Selecionados"
    $btnRun.Location = New-Object System.Drawing.Point(380, 415)
    $btnRun.Size = New-Object System.Drawing.Size(180, 35)
    $btnRun.Add_Click({
        foreach ($nome in $listBox.CheckedItems) {
            $alvo = $apps | Where-Object { $_.DisplayName -eq $nome } | Select-Object -First 1
            if ($alvo) {
                Write-Log "Iniciando remoção: $nome"
                $cmd = $alvo.UninstallString
                if ($cmd -match '^"([^"]+)"(.*)$') {
                    Start-Process -FilePath $Matches[1] -ArgumentList "$($Matches[2]) /quiet" -Wait -NoNewWindow
                } else {
                    $parts = $cmd -split ' ', 2
                    Start-Process -FilePath $parts[0] -ArgumentList $parts[1] -Wait -NoNewWindow
                }
            }
        }
        $form.Close()
    })
    $form.Controls.Add($btnRun)
    $form.ShowDialog() | Out-Null
    Set-Status "Pronto"
})

$BtnReparar.Add_Click({
    Set-Status "Reparando Sistema (SFC/DISM)..."
    Write-Log "Iniciando verificação de integridade..."
    Start-Job -ScriptBlock {
        sfc /scannow
        Dism /Online /Cleanup-Image /RestoreHealth
    } | Out-Null
    Write-Log "Verificação em segundo plano iniciada."
    Set-Status "Pronto"
})

$BtnHardware.Add_Click({
    $os = Get-CimInstance Win32_OperatingSystem
    $cpu = Get-CimInstance Win32_Processor
    $ram = Get-CimInstance Win32_PhysicalMemory | Measure-Object -Property Capacity -Sum
    
    Write-Log "Sistema Operacional: $($os.Caption)"
    Write-Log "Processador: $($cpu.Name)"
    Write-Log "Memória RAM Total: $([math]::round($ram.Sum / 1GB, 2)) GB"
})
