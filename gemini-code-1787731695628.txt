#Requires -RunAsAdministrator

# Configurações via Variáveis de Ambiente (Segurança)
$PortaArmadilha = 8080
$PastaLog       = "$env:USERPROFILE\Desktop\Logs_Trapper"
$ArquivoLog     = "$PastaLog\relatorio_ataques.txt"
$TokenTelegram  = $env:TELEGRAM_TRAPPER_TOKEN
$ChatID         = $env:TELEGRAM_TRAPPER_CHATID

if (-not (Test-Path $PastaLog)) { New-Item -ItemType Directory -Path $PastaLog | Out-Null }

function Enviar-Telegram ($mensagem) {
    if (-not $TokenTelegram -or -not $ChatID) { return }
    try {
        $Url = "https://api.telegram.org/bot$TokenTelegram/sendMessage"
        $Corpo = @{ chat_id = $ChatID; text = $mensagem; parse_mode = "Markdown" }
        Invoke-RestMethod -Uri $Url -Method Post -Body $Corpo -TimeoutSec 5 | Out-Null
    } catch {
        Write-Host "[!] Erro ao enviar notificacao ao Telegram: $_" -ForegroundColor Yellow
    }
}

function Monitorar-TempoReal {
    Clear-Host
    Write-Host "==========================================================" -ForegroundColor Green
    Write-Host "   MONITORAMENTO ATIVO - PORTA ISCA $PortaArmadilha" -ForegroundColor Green
    Write-Host "==========================================================" -ForegroundColor Green
    Write-Host " STATUS  : Aguardando invasões de rede..." -ForegroundColor Green
    Write-Host " TELEGRAM: Notificações ativas no celular." -ForegroundColor Cyan
    Write-Host " COMANDO : Pressione 'M' para voltar ao menu principal..." -ForegroundColor Gray
    Write-Host "----------------------------------------------------------" -ForegroundColor Gray

    while ($true) {
        if ([console]::KeyAvailable) {
            if (([console]::ReadKey($true)).Key -eq 'M') { break }
        }
        
        $novosEventos = Get-WinEvent -LogName Security -FilterXPath "*[System[EventID=5157]]" -MaxEvents 5 -ErrorAction SilentlyContinue
        foreach ($evento in $novosEventos) {
            $xml = [xml]$evento.ToXml()
            $portaAlvo = ($xml.Event.EventData.Data | Where-Object { $_.Name -eq "DestPort" }).'#text'
            $ipInvasor = ($xml.Event.EventData.Data | Where-Object { $_.Name -eq "SourceAddress" }).'#text'
            
            if ($portaAlvo -eq $PortaArmadilha -and $ipInvasor -ne "127.0.0.1" -and $ipInvasor -ne "::1") {
                [console]::Beep(1600, 400)
                $NomeRegra = "TRAPPER_BLOCK_$ipInvasor"
                if (-not (Get-NetFirewallRule -Name $NomeRegra -ErrorAction SilentlyContinue)) {
                    New-NetFirewallRule -DisplayName $NomeRegra -Name $NomeRegra -Direction Inbound -Action Block -RemoteAddress $ipInvasor | Out-Null
                    
                    $DataHora = Get-Date -Format "dd/MM/yyyy HH:mm:ss"
                    "[$DataHora] IP BANIDO: $ipInvasor" | Add-Content -Path $ArquivoLog
                    Write-Host "[🚨] INVASOR BLOQUEADO: $ipInvasor" -ForegroundColor Red
                    
                    Enviar-Telegram "🚨 *Trapper Alerta* 🚨`n`nIP Banido: $ipInvasor"
                }
            }
        }
        Start-Sleep -Seconds 1
    }
}

function Listar-Bloqueados {
    Clear-Host
    Write-Host "=== IPS BLOQUEADOS NO FIREWALL ===" -ForegroundColor Yellow
    $regras = Get-NetFirewallRule -Name "TRAPPER_BLOCK_*" -ErrorAction SilentlyContinue
    if (-not $regras) {
        Write-Host "Nenhum IP bloqueado ainda nesta máquina." -ForegroundColor Gray
    } else {
        foreach ($Regra in $regras) {
            Write-Host " -> $($Regra.Name.Replace('TRAPPER_BLOCK_',''))" -ForegroundColor White
        }
    }
    Write-Host "`nPressione qualquer tecla para voltar..." -ForegroundColor Gray
    [console]::ReadKey($true) | Out-Null
}

while ($true) {
    $host.UI.RawUI.WindowTitle = "TRAPPER v5.3"
    Clear-Host
    Write-Host "==========================================================" -ForegroundColor Cyan
    Write-Host "            TRAPPER v5.3 - CENTRAL DE CONTROLE            " -ForegroundColor Cyan
    Write-Host "==========================================================" -ForegroundColor Cyan
    Write-Host " [1] MONITORAR EM TEMPO REAL" -ForegroundColor White
    Write-Host " [2] VER HISTÓRICO DE IPS BANIDOS" -ForegroundColor White
    Write-Host " [3] SAIR DO SISTEMA" -ForegroundColor White
    Write-Host "----------------------------------------------------------" -ForegroundColor Gray
    
    $Opcao = Read-Host "Escolha uma opção (1-3)"
    if ($Opcao -eq "1") { Monitorar-TempoReal }
    if ($Opcao -eq "2") { Listar-Bloqueados }
    if ($Opcao -eq "3") { exit }
}
